// Create Collection and Insert Data
db.students.insertMany([
  { rollno: 1, name: "Rahul", course: "BCA", marks: 85 },
  { rollno: 2, name: "Amit", course: "BCA", marks: 75 },
  { rollno: 3, name: "Sneha", course: "MCA", marks: 90 },
  { rollno: 4, name: "Priya", course: "MCA", marks: 80 }
]);

// --------------------
// AGGREGATION

// 1. Average marks by course
db.students.aggregate([
  {
    $group: {
      _id: "$course",
      averageMarks: { $avg: "$marks" }
    }
  }
]);

// 2. Total students in each course
db.students.aggregate([
  {
    $group: {
      _id: "$course",
      totalStudents: { $sum: 1 }
    }
  }
]);

// 3. Students with marks > 80
db.students.aggregate([
  {
    $match: { marks: { $gt: 80 } }
  }
]);

// --------------------
// INDEXING

// Create index on name
db.students.createIndex({ name: 1 });

// Create compound index
db.students.createIndex({ course: 1, marks: -1 });

// Show indexes
db.students.getIndexes();

// Drop index
db.students.dropIndex({ name: 1 });
